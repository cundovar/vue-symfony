Js : 

 <script>

// Quand la page a fini de charger
document.addEventListener('DOMContentLoaded', function () { // Récupère tout bouton "Ajouter" du formulaire
let boutonsAjouter = document.querySelectorAll('.add_item_link');

// Pour chaque bouton "Ajouter"
boutonsAjouter.forEach(function (bouton) {
bouton.addEventListener('click', function () { // Cible la bonne liste de blocs
let liste = document.querySelector('.' + bouton.dataset.collectionHolderClass);

// Crée un nouvel élément de la liste
let nouvelBloc = document.createElement('li');
// Utilise le prototype pour générer le HTML
nouvelBloc.innerHTML = liste.dataset.prototype.replace(/__name__/g, liste.dataset.index);

// Ajoute le bouton supprimer
ajouterSupprimer(nouvelBloc);

// Mets le bloc dans la liste
liste.appendChild(nouvelBloc);

// Incrémente l’index du formulaire
liste.dataset.index ++;
  });
});

// Pour chaque bloc déjà présent, ajoute aussi le bouton supprimer
document.querySelectorAll('.pepperLocations li, .pepperCountries li').forEach(function (item) {
ajouterSupprimer(item);
});

// Fonction pour créer le bouton supprimer
function ajouterSupprimer(item) {
let boutonSupprimer = document.createElement('button');
boutonSupprimer.type = 'button';
boutonSupprimer.className = 'btn btn-sm btn-danger ms-2';
boutonSupprimer.innerText = 'Supprimer';

boutonSupprimer.addEventListener('click', function () {
item.remove(); // Supprime le bloc du formulaire
});

item.appendChild(boutonSupprimer);
}
});

</script>
